cmake_minimum_required(VERSION 3.22)
project(futo_opencl C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL ${DIAG_GPU} CACHE BOOL "" FORCE)
set(GGML_OPENCL_PROFILING OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_BIN_KERNELS OFF CACHE BOOL "" FORCE)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fvisibility=hidden")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fvisibility=hidden -fvisibility-inlines-hidden")
# A static shim satisfies FindOpenCL without a startup dependency on the vendor driver.
if(DIAG_GPU)
 add_library(optional_opencl STATIC "${GENERATED}/shim.cpp")
 target_include_directories(optional_opencl PRIVATE "${HEADERS}")
 set(OpenCL_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/liboptional_opencl.a" CACHE FILEPATH "" FORCE)
 set(OpenCL_INCLUDE_DIR "${HEADERS}" CACHE PATH "" FORCE)
endif()
option(FUTO_MIRROR_HOST_TEST "Exercise production mirror in host JNI tests" OFF)
add_subdirectory("${WHISPER_SOURCE}" modern)
if(FUTO_MIRROR_HOST_TEST)
 if(ANDROID OR DIAG_GPU)
  message(FATAL_ERROR "FUTO_MIRROR_HOST_TEST is host CPU testing only")
 endif()
 target_compile_definitions(whisper PRIVATE FUTO_MIRROR_TEST=1)
endif()
if(DIAG_GPU)
 target_link_libraries(ggml-opencl PRIVATE optional_opencl dl)
endif()
add_library(${DIAG_NAME} SHARED bridge.cpp capability.cpp)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
target_link_libraries(${DIAG_NAME} PRIVATE whisper dl)
if(DIAG_GPU)
 # v44 metadata-only capability probe: shares the safe futo_opencl_available
 # dlsym check and the pinned CL headers; it does not link ggml init code.
 target_link_libraries(${DIAG_NAME} PRIVATE optional_opencl)
 target_include_directories(${DIAG_NAME} PRIVATE "${HEADERS}")
endif()
if(ANDROID)
 target_link_libraries(${DIAG_NAME} PRIVATE log)
else()
 target_include_directories(${DIAG_NAME} PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/linux")
endif()
target_compile_definitions(${DIAG_NAME} PRIVATE DIAG_GPU=$<BOOL:${DIAG_GPU}>)
target_link_options(${DIAG_NAME} PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
