cmake_minimum_required(VERSION 3.22)
project(cookie_whistle_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only official Android ARM64 archive is validated")
  endif()
  set(PLATFORM android-arm64)
else()
  set(PLATFORM linux-x86_64)
endif()
set(ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/.cache/${PLATFORM}/libcactus.a")
file(SHA256 "${ARCHIVE}" ARCHIVE_HASH)
if(ANDROID)
  set(EXPECTED 4f75e975f6a859979a2715a89fdbc0fc32742cde001f8674598cf89017d39fdf)
else()
  set(EXPECTED 9e5e7abe1b0bf6675b69a00a5c5e875e80fde496d4e79f7ae86faeae5cab8fd6)
endif()
if(NOT ARCHIVE_HASH STREQUAL EXPECTED)
  message(FATAL_ERROR "Runtime archive hash mismatch")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native/cactus.h" HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.cactus" LICENSE_HASH)
if(NOT HEADER_HASH STREQUAL "facefca17a0d5e715bd92a8b0ea272c1ac34a20a798003b1e691c8fba5ace6e2" OR NOT LICENSE_HASH STREQUAL "cfc7749b96f63bd31c3c42b5c471bf756814053e847c10f3eb003417bc523d30")
  message(FATAL_ERROR "Header/license pin mismatch")
endif()
add_library(cookie_whistle SHARED native/adapter.cpp)
target_compile_options(cookie_whistle PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(cookie_whistle PRIVATE "${ARCHIVE}" m)
if(NOT ANDROID)
  target_link_libraries(cookie_whistle PRIVATE pthread)
endif()
target_link_options(cookie_whistle PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,--exclude-libs,ALL "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")
set_property(TARGET cookie_whistle APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/exports.map")

if(ANDROID)
  target_sources(cookie_whistle PRIVATE native/jni.cpp)
  target_link_options(cookie_whistle PRIVATE -Wl,-z,text)
endif()
option(WHISTLE_TESTS "Build fake-runtime boundary tests on host" OFF)
if(WHISTLE_TESTS AND NOT ANDROID)
  set_property(TARGET cookie_whistle PROPERTY EXCLUDE_FROM_ALL TRUE)
  enable_testing()
  add_executable(adapter_test native/adapter.cpp native/adapter_test.cpp)
  target_compile_options(adapter_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(adapter_test PRIVATE pthread)
  foreach(case good missing-model bad-model cancel-before-load load-error wrong-family native-error unterminated reported-overflow malformed invalid-utf8 nan oversized-input small-output stale-generation cancel-blocked concurrent)
    add_test(NAME ${case} COMMAND adapter_test ${case} "${CMAKE_CURRENT_SOURCE_DIR}/.cache/whistle.cact")
    set_tests_properties(${case} PROPERTIES TIMEOUT 20)
  endforeach()
endif()
add_executable(real_smoke EXCLUDE_FROM_ALL native/adapter.cpp native/smoke.cpp)
target_link_libraries(real_smoke PRIVATE "${ARCHIVE}" m)
if(NOT ANDROID)
  target_link_libraries(real_smoke PRIVATE pthread)
endif()
if(WHISTLE_TESTS AND NOT ANDROID)
  add_executable(model_fd_test native/model_fd_test.cpp)
  add_test(NAME model-fd COMMAND model_fd_test "${CMAKE_CURRENT_SOURCE_DIR}/.cache/whistle.cact")
endif()
